# ── PostgreSQL driver — keep SQLAlchemy and the DBAPI in step ────────────────
# SQLAlchemy is held on the 2.0 line deliberately. A bare "postgresql://" URL
# resolves to a different DBAPI depending on the SQLAlchemy major.minor:
#     2.0.x  ->  psycopg2   (PGDialect_psycopg2)
#     2.1.x  ->  psycopg 3  (PGDialect_psycopg)
# DATABASE_URL uses the bare "postgresql://" scheme, so pairing 2.1.x with
# psycopg2-binary alone raises ModuleNotFoundError: No module named 'psycopg'.
# Keeping 2.0.x makes psycopg2-binary — the driver this project has always
# declared — the driver actually used, with no DATABASE_URL change.
# To move to SQLAlchemy 2.1+ later, either declare psycopg[binary] and drop
# psycopg2-binary, or switch DATABASE_URL to "postgresql+psycopg2://".
sqlalchemy==2.0.57
psycopg2-binary==2.9.13
python-dotenv==1.2.4
pydantic==2.14.2
python-multipart==0.0.32
slowapi==0.1.10
